1"""Pure ISBN helpers: clean, validate and convert ISBN-10 and ISBN-13 values. 

2 

3An ISBN-13 has 13 digits and a check digit that uses the weights 1 and 3. An 

4ISBN-10 has 9 digits and a check character (0 to 9, or ``X`` for 10) that uses 

5the weights 10 down to 1. An ISBN-10 converts to the ISBN-13 with the ``978`` 

6prefix and a new check digit. Only an ISBN-13 with the ``978`` prefix converts 

7back to an ISBN-10. 

8 

9This module imports no Home Assistant code. 

10""" 

11 

12from __future__ import annotations 

13 

14ISBN13_PREFIXES = ("978", "979") 

15ISBN10_PREFIX = "978" 

16 

17 

18class IsbnError(ValueError): 

19 """The value is not a valid ISBN.""" 

20 

21 

22def clean(raw: object) -> str: 

23 """Return *raw* with only its digits and an upper-case ``X``. 

24 

25 Spaces, hyphens, quotes and an ``=`` (as in a Goodreads CSV cell) are removed. 

26 A value that is not a string gives an empty string. 

27 """ 

28 if isinstance(raw, int) and not isinstance(raw, bool): 

29 raw = str(raw) 

30 if not isinstance(raw, str): 

31 return "" 

32 return "".join(ch for ch in raw.upper() if ch.isdigit() or ch == "X") 

33 

34 

35def isbn13_check_digit(first12: str) -> str: 

36 """The check digit for the 12 digits *first12*.""" 

37 total = sum(int(ch) * (3 if index % 2 else 1) for index, ch in enumerate(first12)) 

38 return str((10 - total % 10) % 10) 

39 

40 

41def isbn10_check_char(first9: str) -> str: 

42 """The check character (``0`` to ``9`` or ``X``) for the 9 digits *first9*.""" 

43 total = sum(int(ch) * (10 - index) for index, ch in enumerate(first9)) 

44 check = (11 - total % 11) % 11 

45 return "X" if check == 10 else str(check) 

46 

47 

48def is_valid_isbn13(value: str) -> bool: 

49 """Whether *value* is 13 digits with a correct check digit.""" 

50 if len(value) != 13 or not value.isdigit(): 

51 return False 

52 return isbn13_check_digit(value[:12]) == value[12] 

53 

54 

55def is_valid_isbn10(value: str) -> bool: 

56 """Whether *value* is 9 digits and a correct check character.""" 

57 if len(value) != 10 or not value[:9].isdigit(): 

58 return False 

59 return isbn10_check_char(value[:9]) == value[9] 

60 

61 

62def to_isbn13(isbn10: str) -> str: 

63 """The ISBN-13 for a valid ISBN-10.""" 

64 first12 = ISBN10_PREFIX + isbn10[:9] 

65 return first12 + isbn13_check_digit(first12) 

66 

67 

68def to_isbn10(isbn13: str) -> str | None: 

69 """The ISBN-10 for a valid ISBN-13, or None if its prefix is not ``978``.""" 

70 if not isbn13.startswith(ISBN10_PREFIX): 

71 return None 

72 first9 = isbn13[3:12] 

73 return first9 + isbn10_check_char(first9) 

74 

75 

76def normalize(raw: object) -> tuple[str, str | None]: 

77 """Return ``(isbn13, isbn10)`` for an ISBN-10 or an ISBN-13. 

78 

79 Raise :class:`IsbnError` if *raw* is not a valid ISBN. ``isbn10`` is None for 

80 an ISBN-13 that has no ISBN-10 form. 

81 """ 

82 value = clean(raw) 

83 if len(value) == 13: 

84 if not is_valid_isbn13(value) or not value.startswith(ISBN13_PREFIXES): 

85 raise IsbnError(value) 

86 return value, to_isbn10(value) 

87 if len(value) == 10 and is_valid_isbn10(value): 

88 return to_isbn13(value), value 

89 raise IsbnError(value) 

90 

91 

92def try_normalize(raw: object) -> tuple[str | None, str | None]: 

93 """Like :func:`normalize`, but return ``(None, None)`` for a bad value.""" 

94 try: 

95 return normalize(raw) 

96 except IsbnError: 

97 return None, None 

98 

99 

100def is_isbn_barcode(code: object) -> bool: 

101 """Whether the EAN-13 barcode *code* is an ISBN (prefix ``978`` or ``979``).""" 

102 value = clean(code) 

103 return ( 

104 len(value) == 13 

105 and value.startswith(ISBN13_PREFIXES) 

106 and is_valid_isbn13(value) 

107 ) 

108
